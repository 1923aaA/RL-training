# ****************************************************************************
#                       TDOA TRIMSIM Processor
# For esm_radar cooperative positioning, process information from other sensors.

processor TRIMSIM_PROCESSOR WSF_TRIMSIM_PROCESSOR
#   debug
#   debug_level 2
   minimum_detections 2
end_processor

# ****************************************************************************
#                       ESM Detector

antenna_pattern ESM_RADAR_ANTENNA
   constant 0 db
end_antenna_pattern

sensor ESM_SENSOR WSF_ESM_SENSOR
   frame_time              0.1 sec

   slew_mode               fixed

   maximum_range           3 km       //change

   receiver
      antenna_pattern      ESM_RADAR_ANTENNA
      detection_threshold  1 db       //change
      noise_power          -180 dbw
      internal_loss        0 db
   end_receiver

   error_model trimsim_error
      north_position_error_sigma      0.1 ft
      east_position_error_sigma       0.1 ft
      down_position_error_sigma       0.1 ft
      reference_time_error            5.0 ns
      inter_system_time_delay         5.0 ns
      sensor_timing_error             0.0 s
      atmospheric_refraction_residual 0.0
   end_error_model

   frequency_band          2.5  ghz  3.5  ghz
   frequency_band          9.5  ghz  10.5 ghz

   reports_location
   reports_type
   reports_frequency
end_sensor
