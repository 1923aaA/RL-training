   script string JammerType(double aFrequency)
      string jammerType = "";
      if (aFrequency < 500.e6 && aFrequency > 0)
      {
         jammerType = "SOJ_VHF_JAMMER";
      }
#      if ((aFrequency > 2000.e6) && (aFrequency < 4000.e6))

#      }
#      else if (aFrequency > 8000.e6)
#      {
#         jammerType = "SOJ_XBAND_JAMMER";
#      }
      return jammerType;
   end_script

   processor jammer_task_proc WSF_SCRIPT_PROCESSOR 
      update_interval 1 s
      on_update
         foreach (WsfLocalTrack trk in PLATFORM.MasterTrackList())
         {
            if (trk.ContributorOf(PLATFORM, PLATFORM.Sensor("esm")))
            {
               for (int i=0; i<=trk.SignalCount(); i+=1)
               {
                  double freq = trk.SignalFrequency(i);
                  WsfWeapon jammer;
                  string requiredType = JammerType(freq);
                  for (int j=0; j < PLATFORM.WeaponCount(); j = j + 1)
                  {
                     jammer = PLATFORM.WeaponEntry(j);
                     if (((jammer.QuantityRemaining() > 0) || (jammer.MaximumRequestCount() == 1)) 
                     && PLATFORM.WithinFieldOfView(trk, jammer.Name()) && jammer.Type() == requiredType
                     && jammer.CanJam(freq)
                     )
                     {
                        jammer.StartJamming(freq, 20000, trk);
                     }
                  }
               }
            }
         }
      end_on_update
   end_processor

   end_processor
